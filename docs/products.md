function selectFaceplate(id, imagePath, faceplateMat, chassisMat, keysMat, foilMat) {
  // Update image
  var img = document.getElementById('sc32-showcase-image');
  if (img) {
    img.style.opacity = '0.4';
    img.src = imagePath;
    setTimeout(function() {
      img.style.opacity = '1';
    }, 150);
  }

  // Update button active state
  var buttons = document.querySelectorAll('.te-swatch-btn');
  buttons.forEach(function(btn) {
    btn.classList.remove('active');
  });
  var activeBtn = document.getElementById('btn-swatch-' + id);
  if (activeBtn) {
    activeBtn.classList.add('active');
  }

  // Update CMF readout
  var f = document.getElementById('cmf-faceplate');
  var c = document.getElementById('cmf-chassis');
  var k = document.getElementById('cmf-keys');
  var l = document.getElementById('cmf-foil');
  if (f) f.innerHTML = 'FACEPLATE: ' + faceplateMat;
  if (c) c.innerHTML = 'CHASSIS: ' + chassisMat;
  if (k) k.innerHTML = 'KEYS: ' + keysMat;
  if (l) l.innerHTML = 'FOIL: ' + foilMat;

  // Clear thumbnail active state
  var thumbs = document.querySelectorAll('.te-thumb-item');
  thumbs.forEach(function(th) {
    th.classList.remove('active');
  });
}

// Client-Side Dynamic Category Filter
function filterCatalog(category, btn) {
  var buttons = document.querySelectorAll('.te-filter-btn');
  buttons.forEach(function(b) { b.classList.remove('active'); });
  if (btn) btn.classList.add('active');

  var featuredSec = document.getElementById('featured');
  var manipulativesSec = document.getElementById('manipulatives');
  var calculatorsSec = document.getElementById('calculators');
  var accessoriesSec = document.getElementById('accessories');
  var curriculumSec = document.getElementById('curriculum');
  var downloadsSec = document.getElementById('downloads');

  var allSections = [
    { el: featuredSec, cat: 'calculators' },
    { el: manipulativesSec, cat: 'manipulatives' },
    { el: calculatorsSec, cat: 'calculators' },
    { el: accessoriesSec, cat: 'accessories' },
    { el: curriculumSec, cat: 'curriculum' },
    { el: downloadsSec, cat: 'downloads' }
  ];

  allSections.forEach(function(item) {
    if (!item.el) return;
    if (category === 'all' || item.cat === category) {
      item.el.style.display = '';
    } else {
      item.el.style.display = 'none';
    }
  });

  // Smooth scroll into view when a specific category is clicked
  if (category !== 'all') {
    var target = (category === 'calculators' ? featuredSec : document.getElementById(category));
    if (target) {
      target.scrollIntoView({ behavior: 'smooth', block: 'start' });
    }
  }
}
