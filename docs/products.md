function filterCatalog(category, btn) {
  var buttons = document.querySelectorAll('.te-filter-btn');
  buttons.forEach(function(b) { b.classList.remove('active'); });
  if (btn) btn.classList.add('active');

  var featuredSec = document.getElementById('featured');
  var manipulativesSec = document.getElementById('manipulatives');
  var calculatorsSec = document.getElementById('calculators');
  var accessoriesSec = document.getElementById('accessories');
  var curriculumSec = document.getElementById('curriculum');
  var downloadsSec = document.getElementById('downloads');

  var allSections = [
    { el: featuredSec, cat: 'calculators' },
    { el: manipulativesSec, cat: 'manipulatives' },
    { el: calculatorsSec, cat: 'calculators' },
    { el: accessoriesSec, cat: 'accessories' },
    { el: curriculumSec, cat: 'curriculum' },
    { el: downloadsSec, cat: 'downloads' }
  ];

  allSections.forEach(function(item) {
    if (!item.el) return;
    if (category === 'all' || item.cat === category) {
      item.el.style.display = '';
    } else {
      item.el.style.display = 'none';
    }
  });

  // Smooth scroll into view when a specific category is clicked
  if (category !== 'all') {
    var target = (category === 'calculators' ? featuredSec : document.getElementById(category));
    if (target) {
      target.scrollIntoView({ behavior: 'smooth', block: 'start' });
    }
  }
}
